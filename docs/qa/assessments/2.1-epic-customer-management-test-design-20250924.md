```typescript
describe('Vietnamese Phone Number Validation', () => {
  it('should accept valid Vietnamese mobile numbers', async () => {
    const validMobileNumbers = [
      '0901234567',   // Viettel
      '0987654321',   // Vinaphone
      '0912345678',   // Mobifone
      '0934567890',   // Viettel
      '0886543210'    // Vietnamobile
    ];

      const result = await validateVietnamesePhone(phone);
      expect(result.isValid).toBe(true);
      expect(result.type).toBe('mobile');
      expect(result.carrier).toBeDefined();
    }
  });

  it('should accept valid Vietnamese landline numbers', async () => {
    const validLandlineNumbers = [
      '02834567890',  // Ho Chi Minh City
      '02436789012',  // Hanoi
      '02363456789'   // Da Nang
    ];

    for (const phone of validLandlineNumbers) {
      const result = await validateVietnamesePhone(phone);
      expect(result.isValid).toBe(true);
      expect(result.type).toBe('landline');
      expect(result.region).toBeDefined();
    }
  });

  it('should reject invalid phone number formats', async () => {
    const invalidNumbers = [
      '123456789',     // Too short
      '09012345678',   // Too long
      '0801234567',    // Invalid prefix
      'abcd1234567',   // Contains letters
      '0901-234-567'   // Contains hyphens
    ];

    for (const phone of invalidNumbers) {
      const result = await validateVietnamesePhone(phone);
      expect(result.isValid).toBe(false);
      expect(result.error).toBeDefined();
    }
  });

  it('should handle international phone numbers', async () => {
    const internationalNumbers = [
      '+84901234567',  // Vietnam with country code
      '+1234567890',   // US number
      '+86123456789'   // China number
    ];

    for (const phone of internationalNumbers) {
      const result = await validateVietnamesePhone(phone);
      expect(result.isValid).toBe(true);
      expect(result.type).toBe('international');
    }
  });
});
```

```typescript
describe('Customer Auto-Creation', () => {

    // Verify customer was created
    const newCustomer = await findCustomerByPhone(newPhone);
    expect(newCustomer).toBeDefined();
    expect(newCustomer.fullName).toBe('Nguyễn Văn Minh');
    expect(newCustomer.phone).toBe(newPhone);
    expect(newCustomer.createdAt).toBeDefined();
  });

  it('should not create duplicate customers for existing phone numbers', async () => {
    const existingPhone = '0987654321';

    // Create first repair ticket
    await createRepairTicket({
      customer: { fullName: 'Trần Thị Lan', phone: existingPhone },
      deviceInfo: { brand: 'Dell', model: 'Inspiron' },
      problem: 'Laptop không khởi động'
    });

    const initialCustomerCount = await countCustomers();

    // Create second repair ticket with same phone
    await createRepairTicket({
      customer: { fullName: 'Trần Thị Lan', phone: existingPhone },
      deviceInfo: { brand: 'HP', model: 'Pavilion' },
      problem: 'Bàn phím lỗi'
    });

    const finalCustomerCount = await countCustomers();
    expect(finalCustomerCount).toBe(initialCustomerCount);

    // Verify customer has multiple tickets
    const customer = await findCustomerByPhone(existingPhone);
    const tickets = await getCustomerRepairTickets(customer.id);
    expect(tickets.length).toBe(2);
  });
});
```

## 🔒 **SECURITY & PRIVACY TESTS**

#### **Test Case 2.1.S.1: Customer Data Privacy**
```typescript
describe('Customer Data Privacy', () => {
  it('should enforce role-based access to customer information', async () => {
    const customer = await createCustomer({
      fullName: 'Private Customer',
      phone: '0901999888'
    });

    // Staff should see full information
    const staffView = await getCustomerInfo(customer.id, { role: 'staff' });
    expect(staffView.fullName).toBeDefined();
    expect(staffView.phone).toBeDefined();
    expect(staffView.address).toBeDefined();

    // Public view should be limited
    const publicView = await getCustomerInfo(customer.id, { role: 'public' });
    expect(publicView.fullName).toBeUndefined();
    expect(publicView.phone).toMatch(/\*\*\*\*\*\d{4}/); // Masked phone
    expect(publicView.address).toBeUndefined();
  });

  it('should mask sensitive data in public repair lookups', async () => {
    const customer = await createCustomer({
      fullName: 'Sensitive Data Test',
      phone: '0901234567'
    });

    const ticket = await createRepairTicket({
      customerId: customer.id,
      deviceInfo: { brand: 'ASUS', model: 'VivoBook' },
      problem: 'Màn hình không sáng'
    });

    const publicLookup = await publicRepairLookup(ticket.code);
    expect(publicLookup.customer.name).toMatch(/N\*\*\*\*\* V\*\* M\*\*\*/);
    expect(publicLookup.customer.phone).toMatch(/\*\*\*\*\*4567/);
  });
});
```
