describe('Real-Time Status Synchronization', () => {
  it('should synchronize status updates across multiple sessions', async () => {
    // Create ticket and establish multiple sessions
    const ticket = await createTestRepairTicket();
    const session1 = await createUserSession('staff_001');
    const session2 = await createUserSession('staff_002');
    const session3 = await createUserSession('shop_owner');

    // Subscribe all sessions to ticket updates
    const updates1 = [];
    const updates2 = [];
    const updates3 = [];

    session1.subscribe('ticket_updates', ticket.id, (update) => updates1.push(update));
    session2.subscribe('ticket_updates', ticket.id, (update) => updates2.push(update));
    session3.subscribe('ticket_updates', ticket.id, (update) => updates3.push(update));

    // Update status from session 1
    await session1.updateTicketStatus(ticket.id, 'preliminary_inspection');

    // Wait for propagation
    await sleep(500);

    // Verify all sessions received the update
    expect(updates1.length).toBe(1);
    expect(updates2.length).toBe(1);
    expect(updates3.length).toBe(1);

    expect(updates1[0].newStatus).toBe('preliminary_inspection');
    expect(updates2[0].newStatus).toBe('preliminary_inspection');
    expect(updates3[0].newStatus).toBe('preliminary_inspection');
  });

  it('should handle conflict resolution for concurrent updates', async () => {
    const ticket = await createTestRepairTicket();
    await progressTicketToState(ticket, 'preliminary_inspection');

    const session1 = await createUserSession('staff_001');
    const session2 = await createUserSession('staff_002');

    // Attempt concurrent updates to different fields
    const update1Promise = session1.updateTicket(ticket.id, {
      problem: { description: 'Updated by staff 1' }
    });

    const update2Promise = session2.updateTicket(ticket.id, {
      estimatedCompletionDate: new Date('2025-10-01')
    });

    const [result1, result2] = await Promise.all([update1Promise, update2Promise]);

    // Both updates should succeed (different fields)
    expect(result1.success).toBe(true);
    expect(result2.success).toBe(true);

    // Verify final state contains both updates
    const finalTicket = await getRepairTicket(ticket.id);
    expect(finalTicket.problem.description).toBe('Updated by staff 1');
    expect(finalTicket.estimatedCompletionDate).toBeDefined();
  });

  it('should handle connection management and reliability', async () => {
    const ticket = await createTestRepairTicket();
    const session = await createUserSession('staff_001');

    const updates = [];
    session.subscribe('ticket_updates', ticket.id, (update) => updates.push(update));

    // Simulate connection drop
    session.simulateConnectionDrop();

    // Make updates while disconnected
    await updateTicketStatus(ticket.id, 'preliminary_inspection');
    await updateTicketStatus(ticket.id, 'awaiting_repair_plan');

    // Reconnect
    await session.reconnect();

    // Should receive missed updates
    await sleep(1000);
    expect(updates.length).toBeGreaterThanOrEqual(2);
  });
