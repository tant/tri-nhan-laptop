## 📋 **EPIC OVERVIEW**

Epic 2.3 implements the customer-facing public interface that allows customers to look up their repair status while maintaining strict security controls and data privacy. This system provides transparency to customers while protecting sensitive business information.

#### **Test Case 2.3.1.3: Search Performance & Rate Limiting**
```typescript
describe('Public Interface Performance & Security', () => {
  it('should respond quickly to lookup requests', async () => {
    const lookupTimes = [];

    for (let i = 0; i < 10; i++) {
      const startTime = Date.now();

      await publicLookup({
        method: 'ticket_code',
        value: 'LRP-2025-000001'
      });

      const endTime = Date.now();
      lookupTimes.push(endTime - startTime);
    }

    const averageTime = lookupTimes.reduce((a, b) => a + b, 0) / lookupTimes.length;
    expect(averageTime).toBeLessThan(1000); // < 1 second average

    // No lookup should take more than 3 seconds
    expect(Math.max(...lookupTimes)).toBeLessThan(3000);
  });

  it('should implement rate limiting for security', async () => {
    const clientIP = '192.168.1.100';

    // Make rapid requests (should be limited after threshold)
    const requests = [];
    for (let i = 0; i < 20; i++) {
      requests.push(
        publicLookupWithIP({
          method: 'ticket_code',
          value: `LRP-2025-00000${i}`,
          clientIP: clientIP
        })
      );
    }

    const results = await Promise.all(requests);

    // First several requests should succeed
    const successful = results.filter(r => r.success || r.error !== 'Rate limit exceeded');
    expect(successful.length).toBeGreaterThan(5);

    // Later requests should be rate limited
    const rateLimited = results.filter(r => r.error === 'Rate limit exceeded');
    expect(rateLimited.length).toBeGreaterThan(0);
  });

  it('should handle high concurrent lookup volume', async () => {
    // Simulate 100 concurrent users looking up different tickets
    const concurrentRequests = Array.from({ length: 100 }, (_, i) =>
      publicLookup({
        method: 'ticket_code',
        value: `LRP-2025-${String(i).padStart(6, '0')}`
      })
    );

    const startTime = Date.now();
    const results = await Promise.all(concurrentRequests);
    const endTime = Date.now();

    // All requests should complete within reasonable time
    expect(endTime - startTime).toBeLessThan(5000); // < 5 seconds total

    // At least some requests should succeed (for existing tickets)
    const successful = results.filter(r => r.success);
    expect(successful.length).toBeGreaterThan(0);
  });
});
```

#### **Test Case 2.3.S.1: Data Protection Validation**
```typescript
describe('Public Interface Security', () => {
  it('should prevent information enumeration attacks', async () => {
    // Test ticket code enumeration
    const enumerationAttempts = [];
    for (let i = 1; i <= 100; i++) {
      const code = `LRP-2025-${String(i).padStart(6, '0')}`;
      enumerationAttempts.push(
        publicLookup({ method: 'ticket_code', value: code })
      );
    }

    const results = await Promise.all(enumerationAttempts);

    // All non-existent tickets should return generic error
    const notFoundResults = results.filter(r => !r.success);
    for (const result of notFoundResults) {
      expect(result.error).toBe('Không tìm thấy phiếu sửa chữa với mã này');
      // Should not reveal whether ticket exists or not
      expect(result.details).toBeUndefined();
    }
  });

  it('should validate input sanitization', async () => {
    const maliciousInputs = [
      '<script>alert("xss")</script>',
      "'; DROP TABLE tickets; --",
      '${7*7}',
      '{{constructor.constructor("alert(1)")()}}',
      '../../../etc/passwd'
    ];

    for (const input of maliciousInputs) {
      const result = await publicLookup({
        method: 'ticket_code',
        value: input
      });

      // Should safely reject invalid input
      expect(result.success).toBe(false);
      expect(result.error).toContain('không đúng định dạng');

      // Response should not contain the malicious input
      expect(JSON.stringify(result)).not.toContain(input);
    }
  });

  it('should implement proper session isolation', async () => {
    // Create two separate lookup sessions
    const session1 = await createPublicSession();
    const session2 = await createPublicSession();

    // Each session should be isolated
    const lookup1 = await session1.lookup('LRP-2025-000001');
    const lookup2 = await session2.lookup('LRP-2025-000002');

    // Verify sessions don't leak information
    expect(session1.getHistory()).not.toContain('LRP-2025-000002');
    expect(session2.getHistory()).not.toContain('LRP-2025-000001');
  });
});
```

```typescript
describe('Public Interface Performance', () => {

      const ticketCode = isValidLookup

    const successfulLookups = results.filter(r => r.success);

    );
```
